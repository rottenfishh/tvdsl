function calc(x: int) returns r: int
    locals a: int, b: int, c: int
    temps  t1: int, t2: int, t3: bool, t4: int, t5: int, t6: int, t7: int, t8: int, t9: int, t10: int
    a := 3
    t1 := a + 4
    b := t1
    t2 := b * 2
    c := t2
    t3 := b < 10
    if t3 goto L1
    goto L2
L1:
    t4 := x * 1
    t5 := t4 + 0
    t6 := t5 + c
    r := t6
    goto L3
L2:
    t7 := x / 0
    r := t7
L3:
    t8 := a - 3
    t9 := t8 * x
    t10 := r - t9
    r := t10
    return r
end
