function gcd(a: int, b: int) returns g: int
    locals x: int, y: int, t: int
    temps  t1: bool, t2: int, t3: int, t4: int
    x := a
    y := b
L1:
    t1 := y != 0
    if t1 goto L2
    goto L3
L2:
    t2 := x / y
    t3 := t2 * y
    t4 := x - t3
    t := t4
    x := y
    y := t
    goto L1
L3:
    g := x
    return g
end

function sum_sq(n: int) returns s: int
    locals i: int
    temps  t1: bool, t2: int, t3: int, t4: int
    s := 0
    i := 1
L1:
    t1 := i <= n
    if t1 goto L2
    goto L3
L2:
    t2 := i * i
    t3 := s + t2
    s := t3
    t4 := i + 1
    i := t4
    goto L1
L3:
    return s
end
